cmake_minimum_required(VERSION 3.15)
project(tinyexr LANGUAGES CXX)

add_library(tinyexr tinyexr.cpp)
target_compile_features(tinyexr PRIVATE cxx_std_11)
target_include_directories(tinyexr PRIVATE "${TINYEXR_SOURCE_DIR}")
set_target_properties(tinyexr PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)

foreach(feature MINIZ PIZ ZFP THREAD OPENMP)
    target_compile_definitions(tinyexr PRIVATE TINYEXR_USE_${feature}=$<BOOL:${TINYEXR_USE_${feature}}>)
endforeach()

if(TINYEXR_USE_MINIZ)
    find_package(miniz CONFIG REQUIRED)
    target_link_libraries(tinyexr PRIVATE miniz::miniz)
else()
    find_package(ZLIB CONFIG REQUIRED)
    target_link_libraries(tinyexr PRIVATE ZLIB::ZLIB)
endif()
if(TINYEXR_USE_ZFP)
    find_package(zfp CONFIG REQUIRED)
    target_link_libraries(tinyexr PRIVATE zfp::zfp)
endif()
if(TINYEXR_USE_THREAD)
    find_package(Threads REQUIRED)
    target_link_libraries(tinyexr PRIVATE Threads::Threads)
endif()
if(TINYEXR_USE_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(tinyexr PRIVATE OpenMP::OpenMP_CXX)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tinyexr-openmp.cmake"
        "include(CMakeFindDependencyMacro)\nfind_dependency(OpenMP COMPONENTS CXX)\nset_property(TARGET tinyexr::tinyexr APPEND PROPERTY INTERFACE_LINK_LIBRARIES OpenMP::OpenMP_CXX)\n")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/tinyexr-openmp.cmake" DESTINATION lib/cmake)
endif()

install(TARGETS tinyexr RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES "${TINYEXR_SOURCE_DIR}/tinyexr.h" DESTINATION include)
